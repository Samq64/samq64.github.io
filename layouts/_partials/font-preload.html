Inter bold and italic are left to the pages that use them. */ -}} {{- $urls := partial "font-urls" . -}} {{- range slice "inter-latin-regular" "jetbrains-mono-latin-regular" "jetbrains-mono-latin-800" -}} {{- $name := . -}} {{- $found := "" -}} {{- range $urls -}} {{- if and (not $found) (strings.Contains . (printf "/%s-" $name)) -}} {{- $found = . -}} {{- end -}} {{- end -}} {{- with $found }} {{- else -}} {{- warnf "No built URL for font %q, so it is not being preloaded." $name -}} {{- end -}} {{- end -}}
