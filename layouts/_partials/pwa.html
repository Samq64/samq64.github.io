{{- /* Switched on by a pwa map in front matter. The manifest and worker are written into the page's own directory, so the app's scope is that directory alone. pwa.icon is a 512 square in assets, framed to be shown whole. Android crops a maskable icon to the middle two thirds, so the padded variant is grown from it: 64px a side pulls the eyes in from 195 of 256 to a safe 156, padded in the theme colour the icon is drawn on. Every URL is hashed, the manifest's included, since an installed app holds an icon by its URL. Each is copied to a name saying what it is first, a hash alone leaving four indistinguishable. */ -}} {{- $page := . -}} {{- $pwa := $page.Params.pwa -}} {{- $short := or $pwa.short_name $page.Title -}} {{- $color := site.Params.themeColor -}} {{- $dir := strings.TrimPrefix "/" $page.Path -}} {{- $icon := resources.Get $pwa.icon -}} {{- if not $icon -}} {{- errorf "Icon not found in assets: %s" $pwa.icon -}} {{- end -}} {{- $padded := $icon.Filter (images.Padding 64 $color) -}} {{- $icons := slice -}} {{- range slice (dict "image" $icon "purpose" "any") (dict "image" $padded "purpose" "maskable") -}} {{- $image := .image -}} {{- $purpose := .purpose -}} {{- range slice 512 192 -}} {{- $size := printf "%dx%d" . . -}} {{- $target := printf "%s/icon-%s-%d.png" $dir $purpose . -}} {{- /* Copied rather than re-encoded when the icon is already drawn at this size. */ -}} {{- $source := $image -}} {{- if ne $image.Width . -}} {{- $source = $image.Resize $size -}} {{- end -}} {{- $sized := resources.Copy $target $source | fingerprint -}} {{- $icons = $icons | append (dict "src" $sized.RelPermalink "sizes" $size "type" "image/png" "purpose" $purpose ) -}} {{- end -}} {{- end -}} {{- $manifest := dict "id" $page.RelPermalink "name" $page.Title "short_name" $short "description" $page.Description "start_url" $page.RelPermalink "scope" $page.RelPermalink "display" "standalone" "background_color" $color "theme_color" $color "icons" $icons -}} {{- $file := resources.FromString (printf "%s/manifest.webmanifest" $dir) (jsonify (dict "indent" " ") $manifest) | fingerprint -}} {{- $worker := resources.FromString (printf "%s/sw.js" $dir) (partial "sw.js" $page) -}} {{- /* iOS scales whatever it is handed and crops to a rounded square, so it shares the 192. */ -}} {{- $apple := index (where (where $icons "purpose" "any") "sizes" "192x192") 0 -}} {{- /* No apple-mobile-web-app-*: iOS reads display and short_name from the manifest since 16.4. */}}
