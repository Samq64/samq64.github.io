{{- /* A page's own stylesheets and scripts, built. Both callers want both lists, so the pipelines are spelled out only here. Bundling folds a script's imports under its hash; the bundles are iife, so page-assets defers them rather than marking them modules. */ -}} {{- $styles := slice -}} {{- range .Params.styles -}} {{- with resources.Get . -}} {{- $styles = $styles | append (. | minify | fingerprint) -}} {{- else -}} {{- errorf "Stylesheet not found in assets: %s" . -}} {{- end -}} {{- end -}} {{- $scripts := slice -}} {{- range .Params.scripts -}} {{- with resources.Get . -}} {{- $scripts = $scripts | append (. | js.Build (dict "minify" true) | fingerprint) -}} {{- else -}} {{- errorf "Script not found in assets: %s" . -}} {{- end -}} {{- end -}} {{- return dict "styles" $styles "scripts" $scripts -}}
